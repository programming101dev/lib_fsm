cmake_minimum_required(VERSION 3.14)
project(p101_fsm_tests C CXX)
enable_testing()

set(P101_API_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/../api-manifest.tsv")
set(P101_UNIT_TEST_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/unit-test-manifest.tsv")
if(NOT EXISTS "${P101_API_MANIFEST}" OR NOT EXISTS "${P101_UNIT_TEST_MANIFEST}")
    message(FATAL_ERROR "Public API and unit-test manifests are required")
endif()
file(STRINGS "${P101_API_MANIFEST}" P101_API_ROWS REGEX "^p101_")
file(STRINGS "${P101_UNIT_TEST_MANIFEST}" P101_UNIT_ROWS REGEX "^p101_")
set(P101_API_FUNCTIONS "")
set(P101_UNIT_FUNCTIONS "")
foreach(row IN LISTS P101_API_ROWS)
    string(REGEX MATCH "^p101_[A-Za-z0-9_]+" function_name "${row}")
    list(APPEND P101_API_FUNCTIONS "${function_name}")
endforeach()
foreach(row IN LISTS P101_UNIT_ROWS)
    string(REGEX MATCH "^p101_[A-Za-z0-9_]+" function_name "${row}")
    list(APPEND P101_UNIT_FUNCTIONS "${function_name}")
endforeach()
set(P101_UNIT_FUNCTIONS_UNIQUE ${P101_UNIT_FUNCTIONS})
list(REMOVE_DUPLICATES P101_UNIT_FUNCTIONS_UNIQUE)
list(LENGTH P101_API_FUNCTIONS P101_API_COUNT)
list(LENGTH P101_UNIT_FUNCTIONS P101_UNIT_COUNT)
list(LENGTH P101_UNIT_FUNCTIONS_UNIQUE P101_UNIT_UNIQUE_COUNT)
if(NOT P101_API_COUNT EQUAL P101_UNIT_COUNT OR NOT P101_UNIT_COUNT EQUAL P101_UNIT_UNIQUE_COUNT)
    message(FATAL_ERROR "Every public API must have exactly one unit-test row")
endif()
foreach(function_name IN LISTS P101_API_FUNCTIONS)
    if(NOT function_name IN_LIST P101_UNIT_FUNCTIONS_UNIQUE)
        message(FATAL_ERROR "Missing unit test for ${function_name}")
    endif()
endforeach()

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(P101_TEST_COVERAGE "Instrument the code under test for coverage" OFF)
set(P101_TEST_COVERAGE_FLAGS "")
if(P101_TEST_COVERAGE)
    set(P101_TEST_COVERAGE_FLAGS --coverage)
endif()

set(P101_LIBS p101_error p101_env p101_tool_event p101_c p101_text)
set(P101_PUBLIC_INCLUDE_DIRS "" CACHE STRING "Extra p101 include dirs")
set(P101_PUBLIC_LINK_DIRS "" CACHE STRING "Extra p101 link dirs")
separate_arguments(P101_PUBLIC_INCLUDE_DIRS_LIST NATIVE_COMMAND "${P101_PUBLIC_INCLUDE_DIRS}")
separate_arguments(P101_PUBLIC_LINK_DIRS_LIST NATIVE_COMMAND "${P101_PUBLIC_LINK_DIRS}")
set(_P101_INC_DIRS ${P101_PUBLIC_INCLUDE_DIRS_LIST} /usr/local/include /opt/homebrew/include /opt/local/include)
set(_P101_LIB_DIRS ${P101_PUBLIC_LINK_DIRS_LIST} /usr/local/lib /usr/local/lib64 /opt/homebrew/lib /opt/local/lib)

set(_P101_RESOLVED "")
foreach(_library IN LISTS P101_LIBS)
    unset(_P101_LIB_${_library} CACHE)
    unset(_P101_LIB_${_library})
    find_library(_P101_LIB_${_library} NAMES ${_library} PATHS ${_P101_LIB_DIRS} NO_DEFAULT_PATH)
    if(NOT _P101_LIB_${_library})
        find_library(_P101_LIB_${_library} NAMES ${_library})
    endif()
    if(NOT _P101_LIB_${_library})
        message(FATAL_ERROR "Required p101 library '${_library}' was not found")
    endif()
    list(APPEND _P101_RESOLVED "${_P101_LIB_${_library}}")
endforeach()

add_library(p101_fsm_under_test STATIC
        "${CMAKE_CURRENT_SOURCE_DIR}/../src/effect.c"
        "${CMAKE_CURRENT_SOURCE_DIR}/../src/fsm.c"
)
target_include_directories(p101_fsm_under_test PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}/../include"
        ${_P101_INC_DIRS}
)
target_compile_definitions(p101_fsm_under_test PUBLIC _POSIX_C_SOURCE=200809L _XOPEN_SOURCE=700 P101_FSM_TESTING=1)
if(APPLE)
    target_compile_definitions(p101_fsm_under_test PUBLIC _DARWIN_C_SOURCE)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(p101_fsm_under_test PUBLIC _GNU_SOURCE)
elseif(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
    target_compile_definitions(p101_fsm_under_test PUBLIC _BSD_SOURCE __BSD_VISIBLE)
endif()
target_compile_options(p101_fsm_under_test PRIVATE ${P101_TEST_COVERAGE_FLAGS})
target_link_libraries(p101_fsm_under_test PUBLIC ${_P101_RESOLVED})

add_executable(test_fsm test_fsm.c)
target_link_libraries(test_fsm PRIVATE p101_fsm_under_test)
target_compile_options(test_fsm PRIVATE ${P101_TEST_COVERAGE_FLAGS})
target_link_options(test_fsm PRIVATE ${P101_TEST_COVERAGE_FLAGS})
add_test(NAME test_fsm COMMAND test_fsm)

add_executable(test_fault_wrappers test_fault_wrappers.c)
target_link_libraries(test_fault_wrappers PRIVATE p101_fsm_under_test)
target_compile_options(test_fault_wrappers PRIVATE ${P101_TEST_COVERAGE_FLAGS})
target_link_options(test_fault_wrappers PRIVATE ${P101_TEST_COVERAGE_FLAGS})
add_test(NAME test_fault_wrappers COMMAND test_fault_wrappers)

add_executable(test_cpp_linkage test_cpp_linkage.cpp)
target_link_libraries(test_cpp_linkage PRIVATE p101_fsm_under_test)
target_compile_options(test_cpp_linkage PRIVATE ${P101_TEST_COVERAGE_FLAGS})
target_link_options(test_cpp_linkage PRIVATE ${P101_TEST_COVERAGE_FLAGS})
add_test(NAME test_cpp_linkage COMMAND test_cpp_linkage)
